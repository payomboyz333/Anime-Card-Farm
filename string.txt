-- PayomboyZ Hub - Stage 2: Luarmor Key Checker (String.txt)
-- URL: https://raw.githubusercontent.com/payomboyz333/Anime-Card-Farm/refs/heads/main/string.txt
-- ==============================================================================

local LuarmorConfig = {
    SavedKeyFile = "PayomboyZ_LuarmorKey.txt",
    LuarmorLoaderURL = "https://api.luarmor.net/files/v4/loaders/d0cd427b9eb4643f56e27412fc23b47c.lua",
    StartURL = "https://raw.githubusercontent.com/payomboyz333/Anime-Card-Farm/refs/heads/main/start.txt"
}

local Players = game:GetService("Players")
local CoreGui = game:GetService("CoreGui")
local LocalPlayer = Players.LocalPlayer

-- รับคีย์จาก Start.txt
local userKey = getgenv and getgenv().PayomboyZ_InputKey or nil
if not userKey or userKey == "" then return end

-- ==============================================================================
-- 🛡️ LUARMOR ERROR WATCHER (เบรคลูป Auto-Load กรณีคีย์ผิด/หมดอายุ)
-- ==============================================================================
-- สคริปต์นี้จะทำงานอยู่เบื้องหลังระหว่างที่ Luarmor เช็คคีย์
-- หาก Luarmor เด้งแจ้งเตือนว่าคีย์ผิด มันจะลบคีย์เก่าทิ้งทันที และเรียก Start.txt เพื่อเปิด UI ใหม่อีกรอบ!

local ErrorWatcherActive = true

task.spawn(function()
    while ErrorWatcherActive and task.wait(0.2) do
        pcall(function()
            local containers = {
                CoreGui, 
                LocalPlayer:WaitForChild("PlayerGui"),
                game:GetService("RobloxGui")
            }
            
            for _, container in ipairs(containers) do
                for _, descendant in ipairs(container:GetDescendants()) do
                    local text = ""
                    if descendant:IsA("TextLabel") or descendant:IsA("TextButton") then
                        text = descendant.Text:lower()
                    elseif descendant.Name then
                        text = descendant.Name:lower()
                    end
                    
                    -- เจอแจ้งเตือนคีย์ผิดของ Luarmor
                    if text:find("auth error") 
                        or text:find("whitelist error") 
                        or text:find("no key found") 
                        or text:find("invalid key") 
                        or text:find("need a script_key") 
                    then
                        ErrorWatcherActive = false -- ปิดลูป
                        
                        -- 1. ทำลาย Error Popup ของ Luarmor ทิ้ง
                        local topGui = descendant
                        while topGui and not topGui:IsA("ScreenGui") and topGui.Parent do
                            topGui = topGui.Parent
                        end
                        if topGui and topGui:IsA("ScreenGui") then
                            topGui:Destroy()
                        end
                        
                        -- 2. ลบคีย์ที่หมดอายุออกจากเครื่อง เพื่อกันไม่ให้เกิด Auto-Load วนลูป!
                        if isfile and isfile(LuarmorConfig.SavedKeyFile) and delfile then
                            delfile(LuarmorConfig.SavedKeyFile)
                        end
                        
                        -- 3. สั่งให้ Start.txt เปิดหน้าต่าง UI อีกครั้ง พร้อมโชว์ Error
                        if getgenv then 
                            getgenv().PayomboyZ_ShowUI = true 
                            getgenv().PayomboyZ_ErrorMsg = "❌ Key ไม่ถูกต้อง หรือหมดอายุ! กรุณากรอก Key ใหม่"
                        end
                        
                        loadstring(game:HttpGet(LuarmorConfig.StartURL))()
                        return
                    end
                end
            end
        end)
    end
end)

-- ==============================================================================
-- 🔑 ส่ง Key เข้าสู่ Luarmor Engine
-- ==============================================================================

if getgenv then getgenv().script_key = userKey end
if getrenv then pcall(function() getrenv().script_key = userKey end) end
_G.script_key = userKey
shared.script_key = userKey
script_key = userKey

-- รัน Luarmor Loader (ถ้าคีย์ถูก มันจะข้ามไปยัง Loader.lua แต่ถ้าผิด Watcher ด้านบนจะจับไว้)
local success, err = pcall(function()
    loadstring(game:HttpGet(LuarmorConfig.LuarmorLoaderURL))()
end)

-- ปิดการทำงานของ Watcher ทันทีที่ Luarmor โหลดเสร็จโดยไม่มี Error หลุดมาใน 2 วินาที
task.delay(2.5, function()
    ErrorWatcherActive = false
end)
