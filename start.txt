local function PassToKeyChecker(userKey)
    if not userKey or userKey == "" then return end
    userKey = userKey:gsub("%s+", "")
    
    if getgenv then getgenv().PayomboyZ_InputKey = userKey end

    -- ==============================================================================
    -- 🛡️ EVENT-DRIVEN LUARMOR ERROR WATCHER (Optimized - CPU 0%)
    -- ==============================================================================
    local ErrorWatcherActive = true
    local connections = {}

    local function cleanupConnections()
        ErrorWatcherActive = false
        for _, conn in ipairs(connections) do
            pcall(function() conn:Disconnect() end)
        end
        table.clear(connections)
    end

    local function checkDescendant(descendant)
        if not ErrorWatcherActive then return end
        pcall(function()
            local textStr = ""
            local nameStr = ""

            if descendant:IsA("TextLabel") or descendant:IsA("TextButton") or descendant:IsA("TextBox") then
                textStr = (descendant.Text or ""):lower()
            end
            nameStr = (descendant.Name or ""):lower()

            local function isError(s)
                return s:find("auth error") 
                    or s:find("whitelist error") 
                    or s:find("no key found") 
                    or s:find("invalid key") 
                    or s:find("need a script_key")
                    or s:find("incorrect key")
                    or s:find("key is invalid")
            end

            -- ดักจับกรณี Luarmor เตะออกจากเกม (Kick) ผ่านระบบ Roblox Prompt
            local isKickPrompt = false
            if descendant:IsA("TextLabel") and (descendant.Name == "ErrorMessage" or descendant.Name == "ErrorTitle") then
                if textStr:find("kick") or textStr:find("auth") or textStr:find("whitelist") or textStr:find("unexpected") then
                    isKickPrompt = true
                end
            end

            if isError(textStr) or isError(nameStr) or isKickPrompt then
                cleanupConnections()

                -- ลบคีย์ที่หมดอายุทิ้งทันที เพื่อกันการโหลดซ้ำซาก
                if isfile and isfile(LuarmorConfig.SavedKeyFile) and delfile then
                    delfile(LuarmorConfig.SavedKeyFile)
                end

                -- ทำลาย Error Popup ทิ้ง (ถ้าเป็น GUI ปกติ)
                if not isKickPrompt then
                    local topGui = descendant
                    while topGui and not topGui:IsA("ScreenGui") and topGui.Parent do
                        topGui = topGui.Parent
                    end
                    if topGui and topGui:IsA("ScreenGui") then
                        topGui.Enabled = false
                        topGui:Destroy()
                    else
                        pcall(function() descendant.Parent:Destroy() end)
                    end

                    -- สั่งให้เปิดหน้า Start.txt ขึ้นมาใหม่ทันที
                    if getgenv then 
                        getgenv().PayomboyZ_ShowUI = true 
                        getgenv().PayomboyZ_ErrorMsg = "❌ คีย์หมดอายุหรือไม่ถูกต้อง! กรุณาใส่คีย์ใหม่"
                    end

                    -- แปะเลข tick() ป้องกันการดึงแคชไฟล์เก่าจาก executor
                    loadstring(game:HttpGet(LuarmorConfig.StartURL .. "?t=" .. tostring(tick())))()
                end
            end
        end)
    end

    local function watchContainer(container)
        pcall(function()
            if not container then return end
            -- ตรวจสอบ UI เดิมที่มีอยู่แล้ว
            for _, descendant in ipairs(container:GetDescendants()) do
                task.spawn(checkDescendant, descendant)
            end
            -- ดักจับเมื่อมี UI เพิ่มเข้ามาใหม่
            local conn = container.DescendantAdded:Connect(checkDescendant)
            table.insert(connections, conn)
        end)
    end

    -- เริ่มดักจับ UI ใน CoreGui และ PlayerGui
    local containers = {}
    if pcall(function() return CoreGui end) then table.insert(containers, CoreGui) end
    if pcall(function() return LocalPlayer:FindFirstChild("PlayerGui") end) and LocalPlayer.PlayerGui then table.insert(containers, LocalPlayer.PlayerGui) end
    pcall(function()
        if gethui then
            local hui = gethui()
            if type(hui) == "table" then
                for _, v in ipairs(hui) do
                    if typeof(v) == "Instance" then table.insert(containers, v) end
                end
            elseif typeof(hui) == "Instance" then
                table.insert(containers, hui)
            end
        end
    end)

    for _, container in ipairs(containers) do
        watchContainer(container)
    end

    -- ==============================================================================
    -- 🧠 ระบบจำคีย์แบบอัตโนมัติ (AutoSave) - ถูกลบชั่วคราวตามที่ผู้ใช้สั่ง
    -- ==============================================================================
    task.spawn(function()
        task.wait(10)
        -- ผ่านไป 10 วินาที ถ้าไม่มี Error ตรวจจับเข้ามา ให้เซฟคีย์ลงเครื่อง
        pcall(function()
            if writefile then writefile(LuarmorConfig.SavedKeyFile, userKey) end
        end)
    end)

    -- ==============================================================================
    -- 🔑 ส่ง Key เข้าสู่ Luarmor Engine
    -- ==============================================================================
    if getgenv then getgenv().script_key = userKey end
    if getrenv then pcall(function() getrenv().script_key = userKey end) end
    _G.script_key = userKey
    shared.script_key = userKey
    script_key = userKey

    pcall(function()
        loadstring(game:HttpGet(LuarmorConfig.LuarmorLoaderURL))()
    end)

    -- สั่งให้ยกเลิกการดักจับเหตุการณ์หลังจากผ่านไป 60 วินาที เพื่อคืนค่าความทรงจำ
    task.delay(60, cleanupConnections)
end

-- ==============================================================================
-- 🚀 SMART AUTO-LOGIN (ระบบกันค้าง & เข้าเกมอัตโนมัติ)
-- ==============================================================================
if getgenv and getgenv().PayomboyZ_ShowUI then
    CreateLoaderUI(getgenv().PayomboyZ_ErrorMsg or "❌ คีย์หมดอายุ! กรุณากรอกใหม่", "")
    getgenv().PayomboyZ_ShowUI = false
    return
end

local savedKey = LoadSavedKey()

if savedKey and savedKey ~= "" then
    -- ==============================================================================
    -- 🚀 INSTANT AUTO-LOGIN & ESCAPE HATCH (ฉลาด ไว และแก้บั๊กค้าง 100%)
    -- ==============================================================================
    
    -- สร้างปุ่ม "ยกเลิกคีย์" (Escape Hatch) ไว้บนสุดของจอ เผื่อกรณี Luarmor ค้าง
    local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")
    local EscapeGui = Instance.new("ScreenGui")
    EscapeGui.Name = "PayomboyZ_Escape"
    EscapeGui.ResetOnSpawn = false
    EscapeGui.ZIndexBehavior = Enum.ZIndexBehavior.Global
    pcall(function() EscapeGui.Parent = CoreGui end)
    if not EscapeGui.Parent then EscapeGui.Parent = PlayerGui end

    local EscBtn = Instance.new("TextButton")
    EscBtn.Size = UDim2.new(0, 160, 0, 32)
    EscBtn.Position = UDim2.new(0.5, -80, 0, -40) -- ซ่อนไว้ด้านบนก่อน
    EscBtn.BackgroundColor3 = Color3.fromRGB(255, 68, 93)
    EscBtn.Text = "❌ ลบคีย์เดิม (Reset Key)"
    EscBtn.Font = Enum.Font.GothamBold
    EscBtn.TextSize = 12
    EscBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    EscBtn.BorderSizePixel = 0
    EscBtn.Parent = EscapeGui

    local EscCorner = Instance.new("UICorner")
    EscCorner.CornerRadius = UDim.new(0, 6)
    EscCorner.Parent = EscBtn

    local EscStroke = Instance.new("UIStroke")
    EscStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    EscStroke.Color = Color3.fromRGB(255, 255, 255)
    EscStroke.Transparency = 0.2
    EscStroke.Thickness = 1
    EscStroke.Parent = EscBtn

    -- แอนิเมชั่นเลื่อนปุ่มลงมาตรงกลางบน
    task.spawn(function()
        local ts = game:GetService("TweenService")
        ts:Create(EscBtn, TweenInfo.new(0.5, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Position = UDim2.new(0.5, -80, 0, 20)}):Play()
    end)

    EscBtn.MouseButton1Click:Connect(function()
        -- ถ้าลูกค้ากดปุ่มนี้ แปลว่ามันค้างหรือต้องการเปลี่ยนคีย์
        if isfile and isfile(LuarmorConfig.SavedKeyFile) and delfile then
            delfile(LuarmorConfig.SavedKeyFile)
        end
        EscBtn.Text = "✅ ลบคีย์แล้ว! กำลังเปิดหน้าต่างใหม่..."
        EscBtn.BackgroundColor3 = Color3.fromRGB(50, 200, 50)
        task.wait(1.5)
        
        -- ทำลายปุ่มทิ้ง
        if EscapeGui then EscapeGui:Destroy() end
        
        -- เรียกหน้า Start.txt ขึ้นมาใหม่แบบนุ่มนวล โดยไม่ต้องเตะผู้เล่นออกจากเกม
        if getgenv then 
            getgenv().PayomboyZ_ShowUI = true 
            getgenv().PayomboyZ_ErrorMsg = "✅ ลบคีย์เดิมสำเร็จ! กรุณาใส่คีย์ใหม่"
        end
        loadstring(game:HttpGet(LuarmorConfig.StartURL))()
    end)

    -- ทำลายปุ่มทิ้งอัตโนมัติเมื่อผ่านไป 12 วินาที (แปลว่าเข้าเกมสำเร็จแล้ว)
    task.delay(12, function()
        if EscapeGui then
            local ts = game:GetService("TweenService")
            local tw = ts:Create(EscBtn, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {Position = UDim2.new(0.5, -80, 0, -50)})
            tw:Play()
            tw.Completed:Connect(function() pcall(function() EscapeGui:Destroy() end) end)
        end
    end)

    -- ข้ามหน้าต่าง UI แล้วพุ่งไปหา Luarmor ทันที! (ไม่มีหน่วงเวลา)
    ShowToast("🔑 PayomboyZ Hub", "พบ Key เดิม! กำลังเชื่อมต่ออัตโนมัติ...", false)
    PassToKeyChecker(savedKey)
else
    -- ถ้าไม่มีคีย์ ก็สร้าง UI ให้กรอก
    CreateLoaderUI()
end
