local function LoadSavedKey()
    local saved = ""
    pcall(function()
        if isfile and isfile(LuarmorConfig.SavedKeyFile) and readfile then
            saved = readfile(LuarmorConfig.SavedKeyFile)
        end
    end)
    return saved:gsub("%s+", "")
end

local function ShowToast(title, message, isError)

local function PassToKeyChecker(userKey)
    if not userKey or userKey == "" then return end
    userKey = userKey:gsub("%s+", "")
    
    pcall(function()
        if writefile then writefile(LuarmorConfig.SavedKeyFile, userKey) end
    end)
    
    if getgenv then getgenv().PayomboyZ_InputKey = userKey end
    loadstring(game:HttpGet(LuarmorConfig.StringURL))()
end

        if isfile and isfile(LuarmorConfig.SavedKeyFile) and delfile then
            delfile(LuarmorConfig.SavedKeyFile)
        end
        CreateLoaderUI("⚠️ ยกเลิกการเข้าสู่ระบบอัตโนมัติ กรุณากรอก Key ใหม่", "")
    end)

    task.spawn(function()
        for i = 3, 1, -1 do
            if cancelled then return end
            TText.Text = "⏳ พบ Key เดิม! เข้าสู่ระบบใน " .. i .. "..."
            task.wait(1)
        end
        
        if not cancelled then
            AutoGui:Destroy()
            PassToKeyChecker(savedKey)
        end
    end)
else
    CreateLoaderUI()
end
