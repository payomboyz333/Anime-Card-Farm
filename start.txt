-- Safe HTTP Fetch wrapper (supports native game:HttpGet with executor request fallback)
local function safeFetch(url)
    if not url or url == "" then return "" end

    -- ลำดับที่ 1: ใช้ native game:HttpGet เสมอเพื่อไม่ให้โดน Bot Detection
    local ok, content = pcall(game.HttpGet, game, url)
    if ok and type(content) == "string" and #content > 50 and not content:find("automated source fetching") then
        return content
    end

    -- ลำดับที่ 2: ใช้ executor request หาก game:HttpGet มีปัญหา
    local httprequest = (typeof(request) == "function" and request)
                     or (typeof(http_request) == "function" and http_request)
                     or (typeof(syn) == "table" and typeof(syn.request) == "function" and syn.request)

    if httprequest then
        local okReq, res = pcall(function()
            local headers = {}
            -- Only use WinInet User-Agent on Windows PC to prevent mobile Cloudflare bot flags
            if not (UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled) then
                headers["User-Agent"] = "Roblox/WinInet"
            end
            return httprequest({
                Url = url,
                Method = "GET",
                Headers = headers
            })
        end)
        if okReq and res and res.Body and #res.Body > 50 and not res.Body:find("automated source fetching") then
            return res.Body
        end
    end

    return (ok and content) or ""
end

    if _isScriptLaunched then
        warn("[PAYOMBØYZ HUB] Game script is already launched, ignoring duplicate call.")
        return true
    end

    maxRetries = maxRetries or 3
    local lastErr = nil

    -- ส่ง userKey เข้า Environment ของ PAYOMBØYZ สำหรับสคริปต์ลูกข่าย

                getgenv().PayomboyZ_InputKey = userKey
            end
            if _G then _G.PayomboyZ_UserKey = userKey end
        end)
    end

    local isFlowAuth = type(targetUrl) == "string" and targetUrl:find("flowauth.net")
    local originalScriptKey = (getgenv and getgenv().script_key) or _G.script_key or script_key

    -- เคลียร์ script_key ชั่วคราวสำหรับ Keyless FlowAuth
    if isFlowAuth then
        if getgenv then getgenv().script_key = nil end
        _G.script_key = nil
        script_key = nil
    elseif userKey and userKey ~= "" then
        if getgenv then getgenv().script_key = userKey end
        _G.script_key = userKey
        script_key = userKey
    end

    -- ขั้นตอนที่ 1: ดาวน์โหลดซอร์สโค้ด (Retry เฉพาะ Network/HTTP ถ้าโหลดไม่ผ่าน)
    local fetchUrl = targetUrl
    if type(fetchUrl) == "string" and fetchUrl:find("raw.githubusercontent.com") and not fetchUrl:find("%?") then
        fetchUrl = fetchUrl .. "?t=" .. tostring(os.time())
    end

    local source = nil
    for attempt = 1, maxRetries do
        local okHttp, content = pcall(game.HttpGet, game, fetchUrl)
        if okHttp and type(content) == "string" and #content > 50 and not content:find("404: Not Found") and not content:find("automated source fetching") then
            source = content
            break
        else
            local fallback = safeFetch(fetchUrl)
            if type(fallback) == "string" and #fallback > 50 and not fallback:find("404: Not Found") and not fallback:find("automated source fetching") then
                source = fallback
                break
            end
        end

        lastErr = (okHttp and type(content) == "string" and content:find("automated source fetching"))
            and "FlowAuth Cloudflare rate-limit / bot detection triggered"
            or "Network response empty or timed out"

        if attempt < maxRetries then
            warn(string.format("[PAYOMBØYZ HUB] Fetch attempt %d failed: %s — retrying in 0.5s", attempt, tostring(lastErr)))
            task.wait(0.5)
        end
    end

    if not source or #source < 50 then
        warn("[PAYOMBØYZ HUB] All download attempts failed: " .. tostring(lastErr))
        if isFlowAuth and originalScriptKey then
            pcall(function()
                if getgenv then getgenv().script_key = originalScriptKey end
                _G.script_key = originalScriptKey
                script_key = originalScriptKey
            end)
        end
        _isAuthenticating = false
        if type(onFailCallback) == "function" then
            onFailCallback(lastErr or "Download failed")
        end
        return false
    end

    -- ขั้นตอนที่ 2: Compile ซอร์สโค้ด
    local fn, compileErr = loadstring(source, "=PayomboyZ_Game")
    if not fn then
        warn("[PAYOMBØYZ HUB] Script compile error: " .. tostring(compileErr))
        if isFlowAuth and originalScriptKey then
            pcall(function()
                if getgenv then getgenv().script_key = originalScriptKey end
                _G.script_key = originalScriptKey
                script_key = originalScriptKey
            end)
        end
        _isAuthenticating = false
        if type(onFailCallback) == "function" then
            onFailCallback("Compile error: " .. tostring(compileErr))
        end
        return false
    end

    -- ขั้นตอนที่ 3: สั่งรันสคริปต์แบบแยก Thread (รันครั้งเดียวเท่านั้น! ไม่ retry สคริปต์ซ้ำเมื่อมี runtime error)
    _isScriptLaunched = true
    _isAuthenticating = false

    task.spawn(function()
        local runOk, runErr = pcall(fn)
        if not runOk then
            warn("[PAYOMBØYZ HUB] Target script runtime notice: " .. tostring(runErr))
        end
    end)

    return true
end

-- ==============================================================================
-- 🔑 ฟังก์ชันส่ง Key เข้าสู่ FlowAuth Engine (Auto-Login Handler)
-- ==============================================================================
local function PassToKeyChecker(key)
    if not key or key == "" then return end
    key = key:gsub("%s+", "")

    if _isAuthenticating or _isScriptLaunched then
        warn("[PAYOMBØYZ HUB] Auth or script launch already in progress, skipping duplicate call.")
        return
    end
    _isAuthenticating = true

    task.spawn(function()
        local placeId = game.PlaceId
        local gameData = SupportedGames[placeId]
        local targetUrl = gameData and gameData.URL or "https://raw.githubusercontent.com/Payomboyz0028/Yaranikaaaa/refs/heads/main/WaitKey"
        local mapName = gameData and gameData.Name or "Game Hub"

        -- ตรวจสอบตัวรัน Xeno เฉพาะในแมพที่ไม่รองรับ
        if isXenoExecutor() and gameData and gameData.NoXeno then
            _isAuthenticating = false
            ClearSavedKeys()
            CreateLoaderUI("❌ ตัวรัน Xeno ไม่รองรับในแมพ " .. mapName .. " กรุณาใช้ตัวรันอื่น", key)
            return
        end

        local isSuperVIP = IsSuperVIPKey(key)
        local isVVIP = IsVVIPKey(key)
        local isAuthed = false
        local authReason = nil

        if isSuperVIP then
            isAuthed = true
        else
            isAuthed, authReason = VerifyKeyWithLuarmorAPI(key)
        end

        if isAuthed then
            SaveKey(key)

            -- ดึง Target URL จาก KeyEngine ก่อนเป็นอันดับแรกเพื่อความปลอดภัยสูงสุด
            local engine = GetRemoteKeyData()
            local resolvedUrl = nil
            if engine and type(engine.GetGameURL) == "function" then
                resolvedUrl = engine.GetGameURL(placeId, key)
            end

            if resolvedUrl then
                targetUrl = resolvedUrl
            else
                -- Fallback เดิมหาก KeyEngine ยังไม่ได้อัปเดตบน GitHub
                if gameData and gameData.IsVVIP and not isVVIP then
                    targetUrl = "https://raw.githubusercontent.com/Payomboyz0028/Yaranikaaaa/refs/heads/main/WaitKey"
                end
            end

            -- รัน FlowAuth / Loader ปลายทาง พร้อม Error Callback
            task.defer(function()
                RunFlowAuthWithRetry(targetUrl, key, 3, function(errMsg)
                    _isAuthenticating = false
                    _isScriptLaunched = false
                    task.spawn(function()
                        CreateLoaderUI(L("retryFailed"), key)
                    end)
                end)
            end)
        else
            _isAuthenticating = false
            ClearSavedKeys()
            CreateLoaderUI(authReason or "❌ Key เดิมหมดอายุแล้ว กรุณากรอก Key ใหม่", "")
        end
    end)
end

    local function playTween(object, duration, properties, easingStyle, easingDirection, channel)
        local bucket = activeTweens[object]
        if not bucket then
            bucket = {}
            activeTweens[object] = bucket
        end
        local key = channel or "default"
        local previous = bucket[key]
        if previous then previous:Cancel() end
        local tween = TweenService:Create(
            object,
            TweenInfo.new(
                duration,
                easingStyle or Enum.EasingStyle.Quint,
                easingDirection or Enum.EasingDirection.Out
            ),
            properties
        )
        bucket[key] = tween
        tween.Completed:Connect(function()
            if bucket[key] == tween then
                bucket[key] = nil
            end
        end)
        tween:Play()
        return tween
    end

    -- Procedural High-Voltage Crimson Lightning Bolt Strike Function
    local function strikeLightning(targetFrame, numSegments)
        if not targetFrame or not targetFrame.Parent then return end
        local w = targetFrame.AbsoluteSize.X
        local h = targetFrame.AbsoluteSize.Y
        if w <= 20 or h <= 20 then
            w = DESKTOP_WIDTH
            h = SHELL_HEIGHT
        end

        local startX = math.random(30, math.max(50, math.floor(w - 30)))
        local startY = math.random(4, 20)
        local endX = math.clamp(startX + math.random(-240, 240), 25, w - 25)
        local endY = math.random(math.floor(h * 0.45), math.floor(h * 0.94))

        local segCount = numSegments or math.random(6, 9)
        local pts = {}
        table.insert(pts, Vector2.new(startX, startY))
        for i = 1, segCount - 1 do
            local alpha = i / segCount
            local midX = startX + (endX - startX) * alpha + math.random(-42, 42)
            local midY = startY + (endY - startY) * alpha + math.random(-20, 20)
            table.insert(pts, Vector2.new(midX, midY))
        end
        table.insert(pts, Vector2.new(endX, endY))

        -- Optional Jagged Electric Fork Branch
        local forkPts = {}
        if #pts >= 4 and math.random(1, 10) >= 3 then
            local forkStartIndex = math.random(2, #pts - 2)
            local forkOrigin = pts[forkStartIndex]
            local forkEndX = math.clamp(forkOrigin.X + math.random(-120, 120), 20, w - 20)
            local forkEndY = math.clamp(forkOrigin.Y + math.random(35, 110), 35, h - 20)
            local forkMidX = (forkOrigin.X + forkEndX) / 2 + math.random(-25, 25)
            local forkMidY = (forkOrigin.Y + forkEndY) / 2 + math.random(-14, 14)
            table.insert(forkPts, forkOrigin)
            table.insert(forkPts, Vector2.new(forkMidX, forkMidY))
            table.insert(forkPts, Vector2.new(forkEndX, forkEndY))
        end

        local segFrames = {}
        local function renderSegments(pointList, coreColor, strokeColor, thickness, zIdx)
            for i = 1, #pointList - 1 do
                local p1 = pointList[i]
                local p2 = pointList[i + 1]
                local diff = p2 - p1
                local len = diff.Magnitude
                local angle = math.deg(math.atan2(diff.Y, diff.X))
                local mid = (p1 + p2) / 2

                local seg = create("Frame", {
                    Name = "CrimsonBoltSeg",
                    AnchorPoint = Vector2.new(0.5, 0.5),
                    Position = UDim2.fromOffset(mid.X, mid.Y),
                    Size = UDim2.fromOffset(len + 1, thickness or 2.8),
                    Rotation = angle,
                    BackgroundColor3 = coreColor or Color3.fromRGB(255, 245, 252),
                    BorderSizePixel = 0,
                    Active = false,
                    ZIndex = zIdx or 95,
                }, targetFrame)

                create("UIStroke", {
                    Color = strokeColor or Color3.fromRGB(255, 18, 48),
                    Thickness = 3.2,
                    Transparency = 0.05,
                }, seg)

                table.insert(segFrames, seg)
            end
        end

        -- Render primary bolt & secondary fork
        renderSegments(pts, Color3.fromRGB(255, 248, 252), Color3.fromRGB(255, 18, 48), 3.2, 95)
        if #forkPts > 1 then
            renderSegments(forkPts, Color3.fromRGB(255, 220, 235), Color3.fromRGB(255, 30, 65), 2.2, 94)
        end

        -- Flash ambient crimson glow across entire shell
        if ambientLightningFlash and ambientLightningFlash.Parent then
            ambientLightningFlash.BackgroundTransparency = 0.82
            task.delay(0.05, function()
                if ambientLightningFlash and ambientLightningFlash.Parent then
                    playTween(ambientLightningFlash, 0.18, { BackgroundTransparency = 1 }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
                end
            end)
        end

        -- Multi-stage electric crackle flicker & fade
        task.delay(0.06, function()
            for _, seg in ipairs(segFrames) do
                if seg and seg.Parent then
                    seg.BackgroundTransparency = 0.3
                end
            end
        end)
        task.delay(0.12, function()
            for _, seg in ipairs(segFrames) do
                if seg and seg.Parent then
                    seg.BackgroundTransparency = 0.05
                end
            end
        end)
        task.delay(0.18, function()
            for _, seg in ipairs(segFrames) do
                if seg and seg.Parent then
                    playTween(seg, 0.08, { BackgroundTransparency = 1 })
                end
            end
        end)
        task.delay(0.28, function()
            for _, seg in ipairs(segFrames) do
                if seg and seg.Parent then
                    seg:Destroy()
                end
            end
        end)
    end

    -- 🚀 ExecuteFlowAuth — ทำลาย UI แล้ว Execute พร้อม Retry UI ถ้าล้มเหลว
    -- ==============================================================================
    local function ExecuteFlowAuth(targetUrl, userKey)
        -- Snapshot references ก่อน destroy เพื่อป้องกัน nil access หลัง GUI ถูกทำลาย
        local _gui = gui
        local _conns = trackedConnections
        local _threads = trackedThreads
        local _plansConn = plansMotionConnection
        local _viewConn = viewportConn

        -- ปิด Connection ทั้งหมดก่อน destroy
        pcall(function()
            if _plansConn then _plansConn:Disconnect() end
            if _viewConn then _viewConn:Disconnect() end
            for _, conn in ipairs(_conns) do
                pcall(function() conn:Disconnect() end)
            end
            for _, th in ipairs(_threads) do
                pcall(task.cancel, th)
            end
        end)

        -- ทำลาย UI อย่างปลอดภัย
        pcall(function()
            if _gui and _gui.Parent then
                _gui:Destroy()
            end
            _ENV.__PAYOMBOYZ_CURRENT_GUI = nil
        end)

        -- รัน FlowAuth ใน deferred thread
        task.defer(function()
            task.wait(0.15)
            RunFlowAuthWithRetry(targetUrl, userKey, 3, function(errMsg)
                _isAuthenticating = false
                _isScriptLaunched = false
                task.spawn(function()
                    task.wait(0.2)
                    if getgenv then
                        getgenv().PayomboyZ_ShowUI = true
                        getgenv().PayomboyZ_ErrorMsg = L("retryFailed")
                    end
                    CreateLoaderUI(L("retryFailed"), userKey)
                end)
            end)
        end)
    end

    local function redeemKey()
        if _isAuthenticating or _isScriptLaunched then
            warn("[PAYOMBØYZ HUB] Auth or script launch already in progress, skipping click.")
            return
        end

        local placeId = game.PlaceId
        local gameData = SupportedGames[placeId]
        local mapName = gameData and gameData.Name or "Waiting Hub"
        local isVVIPMap = gameData and gameData.IsVVIP or false

        -- ตรวจสอบตัวรัน Xeno เฉพาะในแมพที่ไม่รองรับ
        if isXenoExecutor() and gameData and gameData.NoXeno then
            setStatus("❌ ตัวรัน Xeno ไม่รองรับในแมพ " .. mapName .. " กรุณาใช้ตัวรันอื่น", COLORS.danger)
            return
        end

        local rawKey = input.Text or ""
        local key = rawKey:gsub("%s+", "")
        if key == "" then
            setStatus("⚠️ กรุณากรอก Key ก่อนเข้าใช้งาน", COLORS.warning)
            return
        end

        _isAuthenticating = true
        setButtonEnabled(redeem, false)

        task.spawn(function()
            -- 1. ตรวจสอบแมพที่กำลังเล่น
            local targetUrl = gameData and gameData.URL or "https://raw.githubusercontent.com/Payomboyz0028/Yaranikaaaa/refs/heads/main/WaitKey"

            -- 2. ตรวจสอบสิทธิ์ SuperVIP / VVIP / Luarmor
            local isSuperVIP = IsSuperVIPKey(key)
            local isVVIP = IsVVIPKey(key)
            local isAuthed = false
            local luarmorReason = nil

            if isSuperVIP then
                setStatus(L("superVIPOk"), COLORS.cyan)
                isAuthed = true
                task.wait(0.15)
            else
                setStatus(L("verifying"), COLORS.cyan)
                isAuthed, luarmorReason = VerifyKeyWithLuarmorAPI(key)
            end

            -- 3. ตรวจสอบว่าคีย์ผ่านไหม
            if isAuthed then
                local engine = GetRemoteKeyData()
                local resolvedUrl = nil
                if engine and type(engine.GetGameURL) == "function" then
                    resolvedUrl = engine.GetGameURL(game.PlaceId, key)
                end

                if resolvedUrl then
                    targetUrl = resolvedUrl
                else
                    if isVVIPMap and not isVVIP then
                        setStatus(L("vvipMapWarn"), COLORS.warning)
                        targetUrl = "https://raw.githubusercontent.com/Payomboyz0028/Yaranikaaaa/refs/heads/main/WaitKey"
                        task.wait(0.8)
                    end
                end

                SaveKey(key)

                if isSuperVIP then
                    setStatus(L("superVIPSuccess") .. mapName, COLORS.success)
                elseif isVVIP then
                    setStatus(L("vvipSuccess") .. mapName, COLORS.success)
                else
                    setStatus(L("authSuccess") .. mapName, COLORS.success)
                end
                task.wait(0.25)

                ExecuteFlowAuth(targetUrl, key)
            else
                _isAuthenticating = false
                setStatus(luarmorReason or L("keyBad"), COLORS.danger)
                setButtonEnabled(redeem, true)
            end
        end)
    end

    local function setDetailsVisible(visible, announce)
        if viewportCompact then
            detailsVisible = false
            userPanel.Size = UDim2.fromOffset(628, 604)
            columnDivider.Visible = false
            if currentCompactView == "main" then
                userPanel.Visible = false
                mainPanel.Position = UDim2.fromOffset(8, 8)
                mainPanel.Visible = true
            else
                mainPanel.Visible = false
                userPanel.Position = UDim2.fromOffset(8, 8)
                userPanel.Visible = true
                if backToKeyBtn then backToKeyBtn.Visible = true end
                if closeInUserBtn then closeInUserBtn.Visible = true end
            end
            setButtonText(settingsButton, "🎮 แมพ & ข้อมูล")
            setButtonTextSize(settingsButton, 10)
            if updateResponsiveScale then
                task.defer(updateResponsiveScale)
            end
            return
        end

        -- Desktop dual-column mode
        setButtonText(settingsButton, "Settings")
        setButtonTextSize(settingsButton, 11)
        if backToKeyBtn then backToKeyBtn.Visible = false end
        if closeInUserBtn then closeInUserBtn.Visible = false end

        if visible == detailsVisible and not detailsTransitioning then
            return
        end

        detailsTransitionVersion = detailsTransitionVersion + 1
        local transitionVersion = detailsTransitionVersion
        detailsTransitioning = true
        detailsVisible = visible

        columnDivider.Visible = true
        if visible then
            userPanel.Size = UDim2.fromOffset(278, 604)
            userPanel.Visible = true
            userPanel.Position = UDim2.fromOffset(-286, 8)
            columnDivider.BackgroundTransparency = 1
            playTween(shell, MOTION.slow, {
                Size = UDim2.fromOffset(DESKTOP_WIDTH, SHELL_HEIGHT),
            }, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut, "layout")
            playTween(mainPanel, MOTION.slow, {
                Position = UDim2.fromOffset(294, 8),
            }, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut, "details")
            playTween(columnDivider, MOTION.slow, {
                BackgroundTransparency = 0.84,
            }, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut, "details")
            local profileTween = playTween(userPanel, MOTION.slow, {
                Position = UDim2.fromOffset(8, 8),
            }, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut, "details")
            profileTween.Completed:Connect(function()
                if transitionVersion == detailsTransitionVersion then
                    detailsTransitioning = false
                end
            end)
        else
            playTween(shell, MOTION.slow, {
                Size = UDim2.fromOffset(COMPACT_WIDTH, SHELL_HEIGHT),
            }, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut, "layout")
            playTween(mainPanel, MOTION.slow, {
                Position = UDim2.fromOffset(8, 8),
            }, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut, "details")
            playTween(columnDivider, MOTION.slow, {
                BackgroundTransparency = 1,
            }, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut, "details")
            local profileTween = playTween(userPanel, MOTION.slow, {
                Position = UDim2.fromOffset(-286, 8),
            }, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut, "details")
            profileTween.Completed:Connect(function()
                if gui.Parent and transitionVersion == detailsTransitionVersion and not detailsVisible then
                    userPanel.Visible = false
                    columnDivider.Visible = false
                    detailsTransitioning = false
                end
            end)
        end
        if announce then
            setStatus(visible and "Profile details shown." or "Compact mode enabled.", COLORS.textMuted)
        end
        if updateResponsiveScale then
            task.defer(updateResponsiveScale)
        end
    end

    settingsButton.Activated:Connect(function()
        if detailsTransitioning then return end
        if viewportCompact then
            if currentCompactView == "main" then
                showCompactView("games")
            else
                showCompactView("main")
            end
            return
        end
        requestedDetails = not requestedDetails
        setDetailsVisible(requestedDetails, true)
    end)

    minimize.Activated:Connect(function()
        if detailsTransitioning then return end
        if viewportCompact then
            shell.Visible = false
            return
        end
        requestedDetails = false
        setDetailsVisible(false, true)
    end)

    local closing = false
    local function closeLoader()
        if closing then return end
        closing = true
        playTween(backdrop, MOTION.normal, { BackgroundTransparency = 1 })
        playTween(shellScale, MOTION.normal, { Scale = shellScale.Scale * 0.96 })
        local tween = playTween(shell, MOTION.normal, { GroupTransparency = 1 }, nil, nil, "visibility")
        tween.Completed:Connect(function()
            if gui.Parent then gui:Destroy() end
        end)
    end

    close.Activated:Connect(closeLoader)
    if closeInUserBtn then
        closeInUserBtn.Activated:Connect(closeLoader)
    end
    redeem.Activated:Connect(redeemKey)
    input.FocusLost:Connect(function(enterPressed)
        if enterPressed then redeemKey() end
    end)

    -- Setup initial Intro Overlay state & Morph Expansion according to Intro.md
    local introOverlay = create("Frame", {
        Name = "IntroOverlay",
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        ZIndex = 999,
    }, shell)

    local introLogo = create("ImageLabel", {
        Name = "IntroLogo",
        Size = UDim2.fromOffset(115, 115),
        Position = UDim2.new(0, 24, 0.5, -57.5),
        BackgroundTransparency = 1,
        Image = loadIntroLogoImage(),
        ImageTransparency = 1,
        ZIndex = 1000,
    }, introOverlay)

    local introTitle = create("TextLabel", {
        Name = "IntroTitle",
        Size = UDim2.new(1, -160, 0, 36),
        Position = UDim2.new(0, 150, 0.5, -26),
        BackgroundTransparency = 1,
        Text = LuarmorConfig.HubName or "PAYOMBØYZ Script HUB",
        TextColor3 = Color3.fromRGB(255, 35, 60),
        Font = Enum.Font.Fondamento,
        TextSize = 28,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextTransparency = 1,
        ZIndex = 1000,
    }, introOverlay)

    create("UIStroke", {
        Color = Color3.fromRGB(80, 0, 15),
        Thickness = 1.8,
    }, introTitle)

    local introSub = create("TextLabel", {
        Name = "IntroSub",
        Size = UDim2.new(1, -160, 0, 18),
        Position = UDim2.new(0, 150, 0.5, 14),
        BackgroundTransparency = 1,
        Text = "LOADING OBSIDIAN V2 ENGINE...",
        TextColor3 = Color3.fromRGB(240, 240, 245),
        Font = Enum.Font.GothamBold,
        TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextTransparency = 1,
        ZIndex = 1000,
    }, introOverlay)

    -- Configure Shell initial intro state (520x150 White Transparent Glass)
    shell.Size = UDim2.fromOffset(520, 150)
    shellChrome.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    shellChrome.BackgroundTransparency = 0.92
    shellStroke.Color = Color3.fromRGB(255, 255, 255)
    shellStroke.Transparency = 0.4
    userPanel.Visible = false
    mainPanel.Visible = false
    columnDivider.Visible = false

    local startedAt = os.clock()
    local sessionThread = task.spawn(function()
        while gui and gui.Parent do
            local elapsed = math.floor(os.clock() - startedAt)
            if sessionText and sessionText.Parent then
                sessionText.Text = string.format("%02d:%02d", math.floor(elapsed / 60), elapsed % 60)
            end
            if pingText and pingText.Parent then
                pingText.Text = "60 FPS"
            end
            task.wait(1)
        end
    end)
    table.insert(trackedThreads, sessionThread)

    local dragging = false
    local dragStart
    local shellStart
    local function onDragStart(inputObject)
        if inputObject.UserInputType == Enum.UserInputType.MouseButton1 or inputObject.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = inputObject.Position
            shellStart = shell.Position
        end
    end
    header.InputBegan:Connect(onDragStart)
    if userPanelDrag then userPanelDrag.InputBegan:Connect(onDragStart) end

    local dragEndedConn = UserInputService.InputEnded:Connect(function(inputObject)
        if inputObject.UserInputType == Enum.UserInputType.MouseButton1 or inputObject.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    local dragChangedConn = UserInputService.InputChanged:Connect(function(inputObject)
        if dragging and (inputObject.UserInputType == Enum.UserInputType.MouseMovement or inputObject.UserInputType == Enum.UserInputType.Touch) then
            local currentScale = (shellScale and shellScale.Scale > 0) and shellScale.Scale or 1
            local delta = (inputObject.Position - dragStart) / currentScale
            shell.Position = UDim2.new(
                shellStart.X.Scale,
                shellStart.X.Offset + delta.X,
                shellStart.Y.Scale,
                shellStart.Y.Offset + delta.Y
            )
        end
    end)

    local entered = false
    local responsiveScale = 1
    updateResponsiveScale = function()
        local camera = workspace.CurrentCamera
        local viewport = camera and camera.ViewportSize or Vector2.new(1280, 720)
        if not viewport or viewport.X < 100 or viewport.Y < 100 then
            viewport = Vector2.new(800, 450)
        end
        local isCompact = viewportCompact or not detailsVisible
        local targetWidth = isCompact and COMPACT_WIDTH or DESKTOP_WIDTH

        local availX = math.max(60, viewport.X - 32)
        local availY = math.max(60, viewport.Y - 32)

        local scaleX = availX / targetWidth
        local scaleY = availY / SHELL_HEIGHT

        responsiveScale = math.clamp(math.min(scaleX, scaleY, 1), 0.20, 1)

        if entered then
            playTween(shellScale, MOTION.normal, { Scale = responsiveScale })
        else
            shellScale.Scale = responsiveScale * 0.96
        end
    end

    local function applyResponsiveLayout()
        local camera = workspace.CurrentCamera
        local viewport = camera and camera.ViewportSize or Vector2.new(1280, 720)
        local isMobile = (deviceName() == "Mobile")
        viewportCompact = isMobile or (viewport.X < 860)
        setDetailsVisible(requestedDetails and not viewportCompact, false)
        updateResponsiveScale()
    end

    applyResponsiveLayout()
    viewportConn = nil
    pcall(function()
        viewportConn = workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(applyResponsiveLayout)
    end)

    gui.Destroying:Connect(function()
        pcall(function()
            if dragEndedConn then dragEndedConn:Disconnect() end
            if dragChangedConn then dragChangedConn:Disconnect() end
            if viewportConn then viewportConn:Disconnect() end
            if plansMotionConnection then plansMotionConnection:Disconnect() end
            for _, conn in ipairs(trackedConnections) do
                pcall(function() conn:Disconnect() end)
            end
            for _, th in ipairs(trackedThreads) do
                pcall(task.cancel, th)
            end
        end)
    end)

    -- Intro Animation Blueprint from Intro.md
    task.spawn(function()
        if not gui.Parent then return end
        entered = true
        playTween(backdrop, MOTION.slow, { BackgroundTransparency = 0.78 })
        playTween(shell, MOTION.slow, { GroupTransparency = 0 }, nil, nil, "visibility")
        playTween(shellScale, MOTION.slow, { Scale = responsiveScale })

        -- Phase 1: Fade-In Logo & Gothic Title (0.45s)
        playTween(introLogo, 0.45, { ImageTransparency = 0 }, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
        playTween(introTitle, 0.45, { TextTransparency = 0 }, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
        playTween(introSub, 0.45, { TextTransparency = 0.2 }, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)

        -- ⚡ STRIKE CRIMSON LIGHTNING ACROSS INTRO CARD
        task.delay(0.18, function()
            pcall(function()
                strikeLightning(shellChrome, 6)
            end)
        end)

        task.wait(1.5)

        -- Phase 2: Fade-Out (0.30s)
        playTween(introLogo, 0.30, { ImageTransparency = 1 }, Enum.EasingStyle.Quart, Enum.EasingDirection.In)
        playTween(introTitle, 0.30, { TextTransparency = 1 }, Enum.EasingStyle.Quart, Enum.EasingDirection.In)
        playTween(introSub, 0.30, { TextTransparency = 1 }, Enum.EasingStyle.Quart, Enum.EasingDirection.In)

        task.wait(0.35)
        introOverlay.Visible = false

        -- Phase 3: Window Morph & Expand (0.65s) to target full UI size
        local isCompact = viewportCompact or not detailsVisible
        local targetWidth = isCompact and COMPACT_WIDTH or DESKTOP_WIDTH
        playTween(shell, 0.65, { Size = UDim2.fromOffset(targetWidth, SHELL_HEIGHT) }, Enum.EasingStyle.Exponential, Enum.EasingDirection.Out)
        playTween(shellChrome, 0.65, { BackgroundColor3 = COLORS.shell, BackgroundTransparency = 0.12 }, Enum.EasingStyle.Exponential, Enum.EasingDirection.Out)
        playTween(shellStroke, 0.65, { Color = Color3.fromRGB(255, 25, 55), Transparency = 0.15 }, Enum.EasingStyle.Exponential, Enum.EasingDirection.Out)

        -- ⚡ HIGH-VOLTAGE MORPH LIGHTNING STRIKE
        task.delay(0.25, function()
            pcall(function()
                strikeLightning(shellChrome, 8)
            end)
        end)

        task.wait(0.65)

        -- Phase 4: Reveal Content Panels
        if isCompact then
            userPanel.Visible = false
            columnDivider.Visible = false
            mainPanel.Position = UDim2.fromOffset(8, 8)
            mainPanel.Visible = true
        else
            userPanel.Size = UDim2.fromOffset(278, 604)
            userPanel.Position = UDim2.fromOffset(8, 8)
            userPanel.Visible = true
            columnDivider.Visible = true
            mainPanel.Position = UDim2.fromOffset(294, 8)
            mainPanel.Visible = true
        end
    end)
end
