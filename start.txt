local Players = game:GetService("Players")
local CoreGui = game:GetService("CoreGui")
local LocalPlayer = Players.LocalPlayer

local function SaveKey(key)
    pcall(function()
        if writefile then writefile(LuarmorConfig.SavedKeyFile, key) end
    end)
end

local function LoadSavedKey()
    local saved = ""
    pcall(function()
        if isfile and isfile(LuarmorConfig.SavedKeyFile) and readfile then
            saved = readfile(LuarmorConfig.SavedKeyFile)
        end
    end)
    return saved:gsub("%s+", "")
end

local function DeleteSavedKey()
    pcall(function()
        if isfile and isfile(LuarmorConfig.SavedKeyFile) and delfile then
            delfile(LuarmorConfig.SavedKeyFile)
        end
    end)
end

-- ==============================================================================
-- 🛡️ COMPREHENSIVE LUARMOR ERROR INTERCEPTOR (Auth Error & Whitelist Error)
-- ดักจับทั้ง "Auth Error", "Whitelist Error", "No key found", "Invalid key" ทั้งหมด
-- ==============================================================================

local isRecoveringFromError = false

local function InterceptLuarmorError(descendant)
    pcall(function()
        if not descendant or isRecoveringFromError then return end
        
        local text = ""
        if descendant:IsA("TextLabel") or descendant:IsA("TextButton") then
            text = descendant.Text:lower()
        elseif descendant.Name then
            text = descendant.Name:lower()
        end
        
        -- ตรวจสอบเงื่อนไขแจ้งเตือนข้อผิดพลาดคีย์ของ Luarmor ทุกรูปแบบ
        if text:find("auth error") 
            or text:find("whitelist error") 
            or text:find("no key found") 
            or text:find("invalid key") 
            or text:find("need a script_key") 
            or text:find("not use it at all") 
        then
            isRecoveringFromError = true
            
            -- หา ScreenGui ตัวแม่ของป็อปอัป Luarmor Error แล้วทำลายทิ้งทันที!
            local topGui = descendant
            while topGui and not topGui:IsA("ScreenGui") and topGui.Parent do
                topGui = topGui.Parent
            end
            
            if topGui and topGui:IsA("ScreenGui") then
                topGui:Destroy()
            end
            
            -- ลบคีย์ที่ผิด/หมดอายุออกจากไฟล์ความจำทันที
            DeleteSavedKey()
            
            -- เด้งป็อปอัปแจ้งเตือน + เรียกหน้าต่างใส่ Key ของเราขึ้นมาใหม่ทันที!
            ShowToast("❌ PayomboyZ Hub", "Key ไม่ถูกต้อง หรือหมดอายุ! กรุณากรอก Key ใหม่", true)
            
            if CreateLoaderUI then
                CreateLoaderUI("❌ Key ไม่ถูกต้อง หรือหมดอายุ! กรุณากรอก Key ใหม่", "")
            end
            
            task.delay(1.5, function()
                isRecoveringFromError = false
            end)
        end
    end)
end

-- ติดตั้ง Listener บนทุก Container ของ UI ใน Roblox ทันที
pcall(function()
    local containers = {
        CoreGui, 
        LocalPlayer:WaitForChild("PlayerGui"),
        game:GetService("RobloxGui")
    }
    
    for _, container in ipairs(containers) do
        container.DescendantAdded:Connect(InterceptLuarmorError)
        for _, desc in ipairs(container:GetDescendants()) do
            InterceptLuarmorError(desc)
        end
    end
end)

-- ฟังก์ชั่นส่ง Key ให้ Luarmor Loader (เพื่อวิ่งต่อไปยัง String.txt)
local function PassKeyToLuarmor(userKey)
    if not userKey or userKey == "" then return end
    userKey = userKey:gsub("%s+", "")
    
    SaveKey(userKey)
    
    -- กำหนดคีย์เข้าสู่ Environment ของ Executor ให้ครบถ้วน
    if getgenv then getgenv().script_key = userKey end
    if getrenv then pcall(function() getrenv().script_key = userKey end) end
    _G.script_key = userKey
    shared.script_key = userKey
    script_key = userKey
    
    -- ส่งคีย์ไปยัง Luarmor Loader (ซึ่งจะปลดล็อกและรัน String.txt)
    script_key = userKey;
    loadstring(game:HttpGet(LuarmorConfig.LuarmorLoaderURL))()
end
